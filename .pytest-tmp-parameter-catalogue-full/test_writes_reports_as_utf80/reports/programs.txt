  Routine: R06_Control [RLL]
    Rung 0 [N]
      Comment: ========== PID DISCHARGE PRESSURE CONTROL ==========

      Logic: NOP();

    Rung 3 [N]
      Comment: PID Error Calculation
      Logic: XIC(PID_Enable)XIO(PT102_Fault)SUB(PID_SP,PID_PV,PID_Err);
    Rung 4 [N]
      Comment: PID Output = Kp*Err + Integral
      Logic: XIC(PID_Enable)XIC(PID_Auto)MUL(PID_Kp,PID_Err,PID_Out)MUL(PID_Ki,PID_Err,ASC_Flow)MUL(ASC_Flow,ScanTime,ASC_Flow)ADD(PID_Int,ASC_Flow,PID_Int)ADD(PID_Out,PID_Int,PID_Out);
    Rung 5 [N]
      Comment: Clamp Output High
      Logic: GRT(PID_Out,CFG_SpeedMax)MOV(CFG_SpeedMax,PID_Out);
    Rung 6 [N]
      Comment: Clamp Output Low

    Rung 7 [N]
      Comment: Anti-Windup: Clamp Integral
      Logic: GRT(PID_Int,50.0)MOV(50.0,PID_Int);
    Rung 8 [N]
      Logic: LES(PID_Int,-50.0)MOV(-50.0,PID_Int);
    Rung 9 [N]
      Comment: ========== ANTI-SURGE CONTROL ==========

      Logic: MOV(FT101_PV,ASC_Flow);
    Rung 10 [N]
      Comment: Calculate Pressure Ratio
      Logic: GRT(PT101_PV,0.1)DIV(PT102_PV,PT101_PV,ASC_PressRatio);
    Rung 11 [N]
      Comment: Simplified Surge Limit Calculation
      Logic: SUB(ASC_PressRatio,1.0,ASC_SurgeLimit)MUL(ASC_SurgeLimit,2000.0,ASC_SurgeLimit)ADD(ASC_SurgeLimit,1000.0,ASC_SurgeLimit);
    Rung 12 [N]
